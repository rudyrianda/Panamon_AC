using System;
using System.Data.SqlClient;

class Program
{
    static void Main()
    {
        string connStr = ""Server=localhost;Database=Panamon_AC;Integrated Security=True;TrustServerCertificate=True;"";
        using (SqlConnection conn = new SqlConnection(connStr))
        {
            conn.Open();
            string sql = ""SELECT SDate, ShiftMode, TotalUnit, MachineCode, Product_Id FROM OEESN WHERE MONTH(SDate) = 7 AND YEAR(SDate) = 2026 AND DAY(SDate) IN (6, 7) ORDER BY SDate"";
            using (SqlCommand cmd = new SqlCommand(sql, conn))
            using (SqlDataReader reader = cmd.ExecuteReader())
            {
                while (reader.Read())
                {
                    Console.WriteLine(reader[""SDate""].ToString() + "" | "" + reader[""ShiftMode""].ToString() + "" | "" + reader[""TotalUnit""].ToString() + "" | "" + reader[""MachineCode""].ToString() + "" | "" + reader[""Product_Id""].ToString());
                }
            }
        }
    }
}
